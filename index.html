from odoo import models, fields, api class HospitalMedicalRecord(models.Model): # Memperluas kapabilitas sistem rekam medis dengan integritas data ketat _name = 'hospital.medical.record' _description = 'Medical Record Transaction' _inherit = ['mail.thread', 'mail.activity.mixin'] patient_id = fields.Many2one('hospital.patient', string='Pasien', required=True, index=True) diagnosis_code = fields.Char(string='Kode ICD-10', tracking=True) state = fields.Selection([ ('draft', 'Draft'), ('verified', 'Terverifikasi'), ('done', 'Selesai') ], default='draft', string='Status Rekam') @api.constrains('diagnosis_code') def _check_diagnosis_validity(self): for record in self: if record.state == 'verified' and not record.diagnosis_code: raise ValidationError("Kode diagnosis wajib diverifikasi sebelum finalisasi.")
